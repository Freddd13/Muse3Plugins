cmake_minimum_required(VERSION 3.16)
project(KumoVstChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt5 REQUIRED COMPONENTS Core Widgets Network Test)
add_subdirectory(../../native/audio/vst vst)
add_library(KumoTestEffect MODULE effect.cpp)
target_include_directories(KumoTestEffect PRIVATE ../../native/thirdparty/kumo-vestige)
target_compile_definitions(KumoTestEffect PRIVATE NOMINMAX)
set_target_properties(KumoTestEffect PROPERTIES PREFIX "")
add_executable(VstChecks checks.cpp)
set_target_properties(VstChecks PROPERTIES AUTOMOC ON)
target_include_directories(VstChecks PRIVATE ../../native)
target_link_libraries(VstChecks PRIVATE kumo_vst Qt5::Test)
target_compile_definitions(VstChecks PRIVATE NOMINMAX QT_NO_KEYWORDS)
if(MSVC)
 target_compile_options(VstChecks PRIVATE /source-charset:utf-8)
endif()
